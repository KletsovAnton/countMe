        // ================= CONFIGS =================

        // Обратите внимание: в коде БОЛЬШЕ НЕТ глобального GEMINI_API_KEY. Он берется из базы для каждого юзера!

        // ===========================================


        const app = initializeApp(firebaseConfig);

        const auth = getAuth(app);

        const db = getFirestore(app);


        let currentUser = null;

        let currentUserApiKey = "";

        let chartInstance = null;

        let allItems = [];

        const categories = ['Еда', 'Вкусняшки (излишество)', 'Средства гигиены', 'Быт', 'Другое'];


        window.handleAuth = async function() {

            const email = document.getElementById('auth-email').value;

            const password = document.getElementById('auth-password').value;

            const errBox = document.getElementById('auth-error');

            errBox.classList.add('hidden');


            try {

                await signInWithEmailAndPassword(auth, email, password);

            } catch (e) {

                try {

                    await createUserWithEmailAndPassword(auth, email, password);

                } catch (err) {

                    errBox.innerText = err.message;

                    errBox.classList.remove('hidden');

                }

            }

        };


        window.logout = function() {

            signOut(auth);

        };


        onAuthStateChanged(auth, async (user) => {

            currentUser = user;

            if (user) {

                document.getElementById('auth-screen').classList.add('hidden');

                document.getElementById('user-email-display').innerText = user.email;


                // Проверяем, есть ли у пользователя сохраненный API-ключ в базе

                const userDocRef = doc(db, `users/${user.uid}`);

                const userDoc = await getDoc(userDocRef);


                if (userDoc.exists() && userDoc.data().geminiApiKey) {

                    currentUserApiKey = userDoc.data().geminiApiKey;

                    document.getElementById('apikey-screen').classList.add('hidden');

                    document.getElementById('app-screen').classList.remove('hidden');

                    loadData();

                } else {

                    // Если ключа нет — показываем окно ввода

                    document.getElementById('apikey-screen').classList.remove('hidden');

                }

            } else {

                document.getElementById('auth-screen').classList.remove('hidden');

                document.getElementById('apikey-screen').classList.add('hidden');

                document.getElementById('app-screen').classList.add('hidden');

            }

        });


        window.saveUserApiKey = async function() {

            const keyInput = document.getElementById('user-gemini-key').value.trim();

            if (!keyInput) {

                alert('Введите ключ!');

                return;

            }


            // Сохраняем ключ в базу данных Firestore в документ пользователя

            await setDoc(doc(db, `users/${currentUser.uid}`), { geminiApiKey: keyInput }, { merge: true });

            currentUserApiKey = keyInput;


            document.getElementById('apikey-screen').classList.add('hidden');

            document.getElementById('app-screen').classList.remove('hidden');

            loadData();

        };


        window.openSettings = function() {

            document.getElementById('user-gemini-key').value = currentUserApiKey;

            document.getElementById('apikey-screen').classList.remove('hidden');

        };


        function fileToBase64(file) {

            return new Promise((resolve, reject) => {

                const reader = new FileReader();

                reader.readAsDataURL(file);

                reader.onload = () => resolve(reader.result.split(',')[1]);

                reader.onerror = error => reject(error);

            });

        }


        window.processReceipt = async function(event) {

            const file = event.target.files[0];

            if (!file) return;


            document.getElementById('loading-box').classList.remove('hidden');


            try {

                const base64Image = await fileToBase64(file);

                

                const prompt = `Распознай все товары с этого чека. Для каждого товара укажи название, цену (число) и распредели строго по одной из категорий: ["Еда", "Вкусняшки (излишество)", "Средства гигиены", "Быт", "Другое"].

                Верни ответ ТОЛЬКО в формате валидного JSON-массива без лишнего текста, например:

                [{"name": "Молоко", "price": 89.50, "category": "Еда"}, {"name": "Чипсы", "price": 120.00, "category": "Вкусняшки (излишество)"}]`;


                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${currentUserApiKey}`, {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify({

                        contents: [{

                            parts: [

                                { text: prompt },

                                { inlineData: { mimeType: file.type, data: base64Image } }

                            ]

                        }]

                    })

                });


                const data = await response.json();

                if (data.error) {

                    throw new Error(data.error.message);

                }


                let textResult = data.candidates[0].content.parts[0].text;

                textResult = textResult.replace(/```json/g, '').replace(/```/g, '').trim();

                const items = JSON.parse(textResult);


                for (let item of items) {

                    await addDoc(collection(db, `users/${currentUser.uid}/items`), {

                        name: item.name,

                        price: Number(item.price),

                        category: item.category,

                        date: new Date().toISOString()

                    });

                }


                loadData();

            } catch (err) {

                alert('Ошибка при распознавании чека: ' + err.message);

            } finally {

                document.getElementById('loading-box').classList.add('hidden');

                event.target.value = '';

            }

        };


        async function loadData() {

            const q = query(collection(db, `users/${currentUser.uid}/items`), orderBy('date', 'desc'));

            const querySnapshot = await getDocs(q);

            allItems = [];

            querySnapshot.forEach((docSnap) => {

                allItems.push({ id: docSnap.id, ...docSnap.data() });

            });


            renderDashboard();

            renderList();

        }


        function renderDashboard() {

            let total = allItems.reduce((sum, item) => sum + (Number(item.price) || 0), 0);

            document.getElementById('total-sum').innerText = total.toFixed(2) + ' ₽';


            let catTotals = {};

            categories.forEach(cat => catTotals[cat] = 0);

            allItems.forEach(item => {

                if (catTotals[item.category] !== undefined) {

                    catTotals[item.category] += Number(item.price) || 0;

                } else {

                    catTotals['Другое'] = (catTotals['Другое'] || 0) + (Number(item.price) || 0);

                }

            });


            const ctx = document.getElementById('expensesChart').getContext('2d');

            if (chartInstance) chartInstance.destroy();


            chartInstance = new Chart(ctx, {

                type: 'doughnut',

                data: {

                    labels: Object.keys(catTotals),

                    datasets: [{

                        data: Object.values(catTotals),

                        backgroundColor: ['#3b82f6', '#ec4899', '#10b981', '#f59e0b', '#64748b']

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    plugins: { legend: { position: 'bottom', labels: { boxWidth: 12, font: { size: 10 } } } }

                }

            });

        }


        function renderList() {

            const listEl = document.getElementById('items-list');

            if (allItems.length === 0) {

                listEl.innerHTML = `<p class="text-xs text-slate-400 text-center py-4">Пока нет сохраненных чеков</p>`;

                return;

            }


            listEl.innerHTML = allItems.map(item => `

                <div class="flex items-center justify-between p-3 bg-slate-50 rounded-xl text-sm">

                    <div class="flex-1 pr-2">

                        <p class="font-medium text-slate-800">${item.name}</p>

                        <div class="flex items-center gap-2 mt-1">

                            <span class="font-bold text-slate-900">${item.price} ₽</span>

                            <select onchange="updateCategory('${item.id}', this.value)" class="text-xs bg-white border border-slate-200 rounded-lg px-2 py-1 text-slate-600 focus:outline-none">

                                ${categories.map(cat => `<option value="${cat}" ${cat === item.category ? 'selected' : ''}>${cat}</option>`).join('')}

                            </select>

                        </div>

                    </div>

                    <button onclick="deleteItem('${item.id}')" class="text-slate-300 hover:text-red-500 p-2"><i class="fa-solid fa-trash-can text-xs"></i></button>

                </div>

            `).join('');

        }


        window.updateCategory = async function(id, newCat) {

            await updateDoc(doc(db, `users/${currentUser.uid}/items`, id), { category: newCat });

            loadData();

        };


        window.deleteItem = async function(id) {

            await deleteDoc(doc(db, `users/${currentUser.uid}/items`, id));

            loadData();

        };
